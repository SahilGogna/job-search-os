% Tailored-resume template used by scripts/generate_resumes.py.
%
% Based on the FAANGPath / Trey Hunner "resume.cls" format (see resume.cls in this same
% directory -- it must ship alongside this file; LaTeX needs it in the same working
% directory to compile). Source: https://github.com/treyhunner/resume
%
% generate_resumes.py fills placeholder tokens below via plain string substitution. Each
% token name, wrapped in double curly braces, appears verbatim exactly ONCE further down
% -- do not repeat a token name anywhere else in this file, including in comments, or it
% will also get substituted. (That is why the list below names them without braces.)
%
%   FULL_NAME          plain text
%   ADDRESS_LINE_1      plain text, phone / city-region-country
%   ADDRESS_LINE_2       plain text, email / links
%   SUMMARY               plain text paragraph
%   SKILLS_BLOCK           complete tabular rows, one or more per skill category
%   EXPERIENCE_BLOCK         LaTeX role blocks, bullets reordered per posting
%   EDUCATION_BLOCK            LaTeX lines, one per degree
%   PROJECTS_BLOCK, LEADERSHIP_BLOCK   optional LaTeX itemize blocks (see below)
%
% SKILLS_BLOCK emits the rows only -- the tabular that wraps them lives here, so layout
% stays in this file. Long categories are wrapped across several rows: the continuation
% rows start with & so the bold label column is empty and the list runs on flush
% underneath. Width and the per-category row cap are controlled by generate_resumes.py
% (max_skill_lines in configs/search.json).
%
% PROJECTS and LEADERSHIP are optional -- each is wrapped in a SECTION marker-comment
% pair (see further down). If context/profile.md has no `projects` or `leadership`
% entries, generate_resumes.py strips the whole marked block rather than leaving
% unfilled placeholder text in a real resume. Do not remove those markers when editing.
%
% Note: resume.cls issues \nofiles, which disables the .aux file hyperref needs -- so
% links are rendered as plain text below, not \href, and hyperref is deliberately not
% loaded here. Adding \href without loading hyperref fails with an undefined control
% sequence.

\documentclass{resume}

\usepackage[left=0.4in,top=0.4in,right=0.4in,bottom=0.4in]{geometry}

\name{{{FULL_NAME}}}
\address{{{ADDRESS_LINE_1}}}
\address{{{ADDRESS_LINE_2}}}

\begin{document}

%----------------------------------------------------------------------------------------
%	PROFILE SUMMARY
%----------------------------------------------------------------------------------------

\begin{rSection}{PROFILE SUMMARY}
{{SUMMARY}}
\end{rSection}

%----------------------------------------------------------------------------------------
%	SKILLS
%----------------------------------------------------------------------------------------

\begin{rSection}{SKILLS}
\begin{tabular}{ @{} >{\bfseries}l @{\hspace{6ex}} l }
{{SKILLS_BLOCK}}
\end{tabular}
\end{rSection}

%----------------------------------------------------------------------------------------
%	EXPERIENCE
%----------------------------------------------------------------------------------------

\begin{rSection}{EXPERIENCE}
{{EXPERIENCE_BLOCK}}
\end{rSection}

%%SECTION:PROJECTS:START%%
%----------------------------------------------------------------------------------------
%	PROJECTS (dropped entirely when the profile has none)
%----------------------------------------------------------------------------------------

\begin{rSection}{PROJECTS}
{{PROJECTS_BLOCK}}
\end{rSection}
%%SECTION:PROJECTS:END%%

%%SECTION:LEADERSHIP:START%%
%----------------------------------------------------------------------------------------
%	LEADERSHIP (dropped entirely when the profile has none)
%----------------------------------------------------------------------------------------

\begin{rSection}{LEADERSHIP}
{{LEADERSHIP_BLOCK}}
\end{rSection}
%%SECTION:LEADERSHIP:END%%

%----------------------------------------------------------------------------------------
%	EDUCATION
%----------------------------------------------------------------------------------------

\begin{rSection}{EDUCATION}
{{EDUCATION_BLOCK}}
\end{rSection}

\end{document}
