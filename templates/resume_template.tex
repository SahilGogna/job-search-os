% Tailored-resume template used by scripts/generate_resumes.py.
%
% Based on the FAANGPath / Trey Hunner "resume.cls" format (see resume.cls in
% this same directory -- it must ship alongside this file; LaTeX needs it in
% the same working directory to compile). Source: https://github.com/treyhunner/resume
%
% generate_resumes.py fills placeholder tokens below via plain string
% substitution (each token name, wrapped in double curly braces, appears
% verbatim exactly once further down in this file -- do not repeat a token
% name elsewhere in this file, including in comments, or it will also get
% substituted):
%
%   FULL_NAME          plain text
%   ADDRESS_LINE_1      plain text, phone / city-region-country
%   ADDRESS_LINE_2       plain text, email / links
%   SUMMARY               plain text paragraph
%   SKILLS_LIST            plain text, comma-separated, reordered per posting
%   EDUCATION_BLOCK          LaTeX lines, one per degree
%   EXPERIENCE_BLOCK           LaTeX rSubsection blocks, bullets reordered per posting
%   PROJECTS_BLOCK, LEADERSHIP_BLOCK   optional LaTeX blocks (see below)
%
% PROJECTS and LEADERSHIP are optional -- each is wrapped in a
% SECTION marker-comment pair (see further down). If context/profile.md has
% no `projects` or `leadership` entries, generate_resumes.py strips the whole
% marked block rather than leaving unfilled placeholder text in a real resume.
% Do not remove those markers when editing this file.

\documentclass{resume}

\usepackage[left=0.4in,top=0.4in,right=0.4in,bottom=0.4in]{geometry}
% Note: resume.cls issues \nofiles, which disables the .aux file hyperref
% needs -- so links are rendered as plain text below, not \href, and
% hyperref is deliberately not loaded here.

\name{{{FULL_NAME}}}
\address{{{ADDRESS_LINE_1}}}
\address{{{ADDRESS_LINE_2}}}

\begin{document}

\begin{rSection}{Objective}
{{SUMMARY}}
\end{rSection}

\begin{rSection}{Education}
{{EDUCATION_BLOCK}}
\end{rSection}

\begin{rSection}{Skills}
\begin{tabular}{ @{} >{\bfseries}l @{\hspace{6ex}} l }
Skills & {{SKILLS_LIST}}
\end{tabular}
\end{rSection}

\begin{rSection}{Experience}
{{EXPERIENCE_BLOCK}}
\end{rSection}

%%SECTION:PROJECTS:START%%
\begin{rSection}{Projects}
{{PROJECTS_BLOCK}}
\end{rSection}
%%SECTION:PROJECTS:END%%

%%SECTION:LEADERSHIP:START%%
\begin{rSection}{Leadership}
{{LEADERSHIP_BLOCK}}
\end{rSection}
%%SECTION:LEADERSHIP:END%%

\end{document}
